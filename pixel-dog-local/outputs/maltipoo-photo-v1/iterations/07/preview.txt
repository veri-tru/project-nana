def union(a,b,k=1.):
    h=np.clip(.5+.5*(b-a)/k,0,1)
    return b*(1-h)+a*h-k*h*(1-h)

def ellipsoid(q,c,r):
    v=(q-np.asarray(c))/r
    k0=np.linalg.norm(v,axis=-1);k1=np.linalg.norm(v/r,axis=-1)
    return k0*(k0-1)/np.maximum(k1,1e-7)

def capsule(q,a,b,r):
    d=b-a;t=np.clip(np.sum((q-a)*d,axis=-1)/np.dot(d,d),0,1)
    return np.linalg.norm(q-a-t[...,None]*d,axis=-1)-r

    if b=='maltipoo':profile['eye_radius']=(1.5,1.5)
    features=model.head_features(rig,b,profile)

        # rebuilt cranium. No separate sheet connector or decorative outline.

        zz=PchipInterpolator(knots,[5.6,7.8,10.0,10.2,9.6])(t)
        width=PchipInterpolator(knots,[0,3.2,4.9,4.4,0])(t)
        depth=PchipInterpolator(knots,[0,1.8,2.3,2.1,0])(t)
        tip=np.sqrt(np.clip(1-((t-.78)/.22)**2,0,1))
        width=np.where(t>.78,4.4*tip,width)
        depth=np.where(t>.78,2.1*tip,depth)

        for sign in (1,-1):
            root=np.array(model.pose(rig)[f'face.{"L" if sign==1 else "R"}_ear_root'])-model.pose(rig)['body.head']
            ear=np.stack(np.broadcast_arrays(xx[:,None]+width[:,None]*np.cos(theta),yy[:,None],sign*(zz[:,None]+depth[:,None]*np.sin(theta))),axis=-1)
            ear+=root-np.array([2.,9.3,sign*7.7])

    if b=='whippet':

        w=1-model.smoothstep((v[:,0,0]+5)/8)
        v[:,:,1]+=(target-lo)[:,None]*w[:,None]*np.clip((hi[:,None]-v[:,:,1])/np.maximum(.01,hi-lo)[:,None],0,1)

    local=raw_distance/np.maximum(magnitude,1.)
    weight=1-model.smoothstep((np.abs(raw_distance)-.35)/.65)
    distance=distance*(1-weight)+local*weight
    s['head_distance']=gaussian_filter(distance,1.2).astype(np.float32)

def lip_height(x,s,b):
    p=PARAMS[b];t=np.clip((x-s['xs'][0]-p['front'])/max(1,-5-s['xs'][0]),0,1)
    return p['lip_front']+(p['lip']-p['lip_front'])*np.sin(t*np.pi/2)

def raw_head_field(q,s,b):
    x,y,z=np.moveaxis(q,-1,0);xs=s['xs'];cy=np.interp(x,xs,s['cy'])
    a=np.mod(np.arctan2(z,y-cy),2*np.pi)
    ix=np.interp(x,xs,np.arange(len(xs)))
    radius=map_coordinates(s['radius'],[ix.ravel(),(a*360/(2*np.pi)).ravel()],order=1,mode='nearest').reshape(x.shape)
    d=np.hypot(y-cy,z)-radius
    d=np.maximum(d,np.maximum(xs[0]-x,x-xs[-1]))
    if b=='maltipoo':
        frame=s['facial_frame'];sk=frame['face.skull'];mx=frame['face.maxilla']
        skull=ellipsoid(q,sk,[frame['face.occiput'][0]-sk[0]+1.5,frame['face.crown'][1]-sk[1]-.5,abs(frame['face.L_temple'][2])+ .6])
        muzzle=ellipsoid(q,mx,[mx[0]-frame['face.upper_lip'][0]-.2,3.6,abs(frame['face.L_muzzle_side'][2])+1.3])
        rebuilt=union(skull,muzzle,3.0)
        # A low rounded whisker pad joins the cheeks; no ring around the eyes.
        whiskers=ellipsoid(q,mx+[-1.,-.4,0],[3.4,2.8,abs(frame['face.L_muzzle_side'][2])+.2])
        rebuilt=union(rebuilt,whiskers,2.0)
        nose=np.maximum(d,x-(xs[0]+2.9))
        # One union joins the retained common nasal cap to the facial volume.
        # Spatially blending unrelated distance fields would create a box rim.
        d=union(rebuilt,nose,.65)
    return d

def head_field(q,s,rig,b):
    x,y,z=np.moveaxis(q,-1,0);xs=s['xs']
    coords=np.moveaxis((q-s['sdf_origin'])/s['sdf_step'],-1,0)
    d=map_coordinates(s['head_distance'],coords.reshape(3,-1),order=1,mode='constant',cval=50).reshape(x.shape)
    world={k:np.array(v) for k,v in model.pose(rig).items()};o=world['body.head']
    ctr=np.array([world[k]-o for k in ['head.occipital_joint','neck.C1','neck.C2','neck.lower']])
    neck=np.full(x.shape,1e3)
    for i in range(3):
        for t in np.linspace(0,1,9):
            c=ctr[i]*(1-t)+ctr[i+1]*t
            progress=(i+t)/3
            # Broad base-of-skull support; taper only slightly toward the cut.
            r=np.array([rig['neck_skin']['depth'],4.4,rig['neck_skin']['width']])*(1-.10*progress)
            neck=union(neck,ellipsoid(q,c,r),.35)
    # The oral roof follows the curved lip, rising into the rear cheek before
    # the cervical mass. It does not stretch down with the rotating jaw.
    hinge=world['face.jaw_axis']-o
    corner=world['face.L_mouth_corner'][0]-o[0]
    rear=model.smoothstep((x-(corner-2.5))/3.0)
    floor=lip_height(x,s,b)
    width=np.interp(x,xs,np.abs(s['modified'][:,:,2]).max(1))*.96
    if b=='maltipoo':width=np.maximum(width,5.1*np.sqrt(np.clip(1-((x+6.6)/4.7)**2,0,1)))
    thick=PARAMS[b]['depth']*.52
    floor=floor-thick+thick*np.sqrt(np.clip(1-(z/np.maximum(width,.1))**2,0,1))
    floor-=PARAMS[b]['depth']*rear
    # The mouth corner is anterior to TMJ. Keep the cheek over the ascending
    # ramus, rather than cutting a triangle all the way up to the hinge.
    throat=model.smoothstep((x-(hinge[0]-2.0))/6.0)
    floor-=np.maximum(2.,PARAMS[b]['lip']-PARAMS[b]['depth']-ctr[-1,1]+1)*throat
    cut=floor-y
    if b=='maltipoo':
        # Split the same rest envelope after cervical union. The lower jaw is
        # extracted from the complementary half of this shared oral boundary.
        roof=union(union(oral_split(q,s),corner+.5-x,.8),3.5-np.abs(z),1.2)
        return np.maximum(-union(-union(d,neck,3.2),-roof,.20),ctr[-1,1]-y)
    d=-union(-d,-cut,.8)
    return np.maximum(union(d,neck,3.2),ctr[-1,1]-y)

def oral_split(q,s):
    x,y,z=np.moveaxis(q,-1,0);f=s['facial_frame']
    lip=f['face.upper_lip'];corner=f['face.L_mouth_corner'];hinge=f['face.jaw_axis']
    u=model.smoothstep((x-lip[0])/max(1,corner[0]-lip[0]))
    height=lip[1]+.8+(corner[1]+.6-lip[1]-.8)*u
    height+=.25*(z/3.5)**2
    return height-y


def maltipoo_jaw(s,rig):
    pose=model.pose(rig);hinge=np.array(pose['face.jaw_axis'])-pose['body.head']
    corner=s['facial_frame']['face.L_mouth_corner'];start=s['xs'][0]+.6;end=hinge[0]-1.2
    x=np.linspace(start,end,100);thick=PARAMS['maltipoo']['depth']*.52
    yy=oral_split(np.column_stack([x,x*0,x*0]),s)-thick
    width=np.minimum(4.8,np.maximum(.1,5*np.sqrt(np.clip(1-((x+6)/4.5)**2,0,1))))
    def field(q):
        xx,y,z=np.moveaxis(q,-1,0)
        coords=np.moveaxis((q-s['sdf_origin'])/s['sdf_step'],-1,0)
        rest=map_coordinates(s['head_distance'],coords.reshape(3,-1),order=1,mode='constant',cval=50).reshape(xx.shape)
        d=np.maximum(rest,-oral_split(q,s)-.015)
        d=-union(-d,-(np.abs(z)-4.0),.6)
        d=-union(-d,-(xx-(corner[0]+.1)),.8)
        for side,sign in [('L',1),('R',-1)]:
            a=np.array([corner[0]-.6,corner[1]-.2,sign*3.5])
            h=np.array(pose[f'face.{side}_jaw_hinge'])-pose['body.head']
            d=union(d,capsule(q,a,h,.75),.5)
        return d
    v,f,n=extract(field,[(start-2,hinge[0]+2),(-5,hinge[1]+2),(-12,12)],.17)
    return dict(v=v,f=f,n=n,hinge=hinge,field=field,x=x,y=yy,width=width,thick=thick)


def extract(field,bounds,step=STEP):
    axes=[np.arange(a,b+step,step) for a,b in bounds]
    q=np.stack(np.meshgrid(*axes,indexing='ij'),axis=-1)
    values=field(q).astype(np.float32)
    v,f,n,_=marching_cubes(values,0,spacing=(step,)*3,allow_degenerate=False)
    v+=np.array([a[0] for a in bounds]);n=-n
    # skimage normals point into a negative-inside field; recompute smooth
    # outward normals from the gradient so shading is independent of winding.
    return v,f,n

def jaw_mesh(s,rig,b):
    if b=='maltipoo':return maltipoo_jaw(s,rig)
    p=PARAMS[b];pose=model.pose(rig);hinge=np.array(pose['face.jaw_axis'])-pose['body.head']
    start=s['xs'][0]+p['front'];end=hinge[0]-1.2
    x=np.linspace(start,end,100)
    width=np.interp(x,s['xs'],np.abs(s['modified'][:,:,2]).max(1))*.86
    if b=='maltipoo':
        width=np.maximum(width,4.0*np.sqrt(np.clip(1-((x+7.5)/3.4)**2,0,1)))
    # Keep the mandibular footprint under the upper muzzle/nasal width in
    # dorsal view; a broad whisker-pad radius is not a lower-jaw radius.
    width=np.minimum(width,{'golden':6.2,'whippet':3.3,'dachshund':5.0,'maltipoo':4.8}[b])
    width*=np.sqrt(np.clip((x-start)/1.1,0,1));width=np.maximum(width,.08)
    corner=pose['face.L_mouth_corner'][0]-pose['body.head'][0]
    narrow=model.smoothstep((x-corner)/max(1,end-corner))
    hinge_width=abs(pose['face.L_jaw_hinge'][2]-pose['body.head'][2])*.82
    width=width*(1-narrow)+hinge_width*narrow
    yy=lip_height(x,s,b)-p['depth']*.52
    thick=p['depth']*.52
    if b=='maltipoo':
        # Seat the mandibular top on the actual closed upper-muzzle underside.
        samples=np.linspace(-4,12,321)
        q=np.stack(np.broadcast_arrays(x[:,None],samples[None,:],np.zeros((len(x),len(samples)))),axis=-1)
        values=head_field(q,s,rig,b)
        first=(values<0).argmax(1)
        first=np.maximum(first,1)
        lo=samples[first-1];hi=samples[first]
        for _ in range(12):
            mid=(lo+hi)/2
            hit=head_field(np.column_stack([x,mid,x*0]),s,rig,b)<0
            hi=np.where(hit,mid,hi);lo=np.where(hit,lo,mid)
        roof=(lo+hi)/2
        taper=np.sqrt(np.clip((x-start)/1.4,0,1))
        fitted=roof-np.maximum(.04,thick*taper)+.02
        w=1-model.smoothstep((x-(corner-2.0))/2.0)
        yy=yy*(1-w)+fitted*w
    def field(q):
        xx,y,z=np.moveaxis(q,-1,0)
        cy=np.interp(xx,x,yy);w=np.interp(xx,x,width)
        taper=np.sqrt(np.clip((xx-start)/1.4,0,1))
        h=np.maximum(.04,thick*taper)
        d=(np.sqrt(((y-cy)/h)**2+(z/w)**2)-1)*h
        d=np.maximum(d,np.maximum(start-xx,xx-end))
        for sign in (1,-1):
            a=np.array([end-.9,yy[-1],sign*width[-1]*.80])
            h=np.array(pose[f'face.{"L" if sign==1 else "R"}_jaw_hinge'])-pose['body.head']
            d=union(d,capsule(q,a,h,p['depth']*.49),.7)
        if b=='maltipoo':
            avoid=np.minimum(-head_field(q,s,rig,b)-.015,(corner-1.-xx)*2.)
            d=np.maximum(d,avoid)
        return d
    v,f,n=extract(field,[(start-2,hinge[0]+2),(-5,hinge[1]+2),(-12,12)],.17)
    return dict(v=v,f=f,n=n,hinge=hinge,field=field,x=x,y=yy,width=width,thick=thick)

def rotate(v,hinge,degrees):
    a=np.radians(degrees);r=np.array([[np.cos(a),-np.sin(a),0],[np.sin(a),np.cos(a),0],[0,0,1]])
    return (v-hinge)@r.T+hinge

def colors(v,s,rig,b,state):
    body=np.array(model.old.COLORS.get(b,([239,228,210],[220,199,172]))[0],float)
    if b=='maltipoo':body=MALTIPOO_COAT.copy()
    c=np.tile(body,(len(v),1));x,y,z=v.T
    # Transfer the original nasal material only, through the retained rostral
    # parameterization. Eye dots use unchanged centers and circular radii.
    ov,oc=s['features'][0]
    centers=(ov[:-1,:-1]+ov[1:,:-1]+ov[1:,1:]+ov[:-1,1:])/4
    if b=='maltipoo': nose_end=s['xs'][0]+2.9
    else:nose_end=s['xs'][0]+4.4*model.NOSE_SCALE[b]
    mask=x<nose_end
    dist,ix=cKDTree(centers.reshape(-1,3)).query(v[mask],k=4)
    weights=1/np.maximum(dist,.02)**2;weights/=weights.sum(1)[:,None]
    c[mask]=(oc.reshape(-1,3)[ix]*weights[:,:,None]).sum(1)
    if b=='maltipoo':
        pigment=model.smoothstep((150-c[mask].max(1))/65)
        pigment*=1-model.smoothstep((dist.min(1)-.15)/.35)
        c[mask]=body*(1-pigment[:,None])+c[mask]*pigment[:,None]
    pose=model.pose(rig);radius=s['profile']['eye_radius'][0]
    for side,sign in [('L',1),('R',-1)]:
        eye=np.array(pose[f'face.{side}_eye'])-pose['body.head']
        chosen=np.argmin((x-eye[0])**2+(y-eye[1])**2+np.where(sign*z>0,0,1e5))
        center=v[chosen];u=(x-eye[0])/radius;w=(y-eye[1])/radius
        ds=np.sum((v-center)**2,axis=1)/radius**2;alpha=np.clip((1.1-ds)*5,0,1)*(sign*z>0)
        if state=='half':alpha*=model.smoothstep((-w+.12)/.20)
        if state=='closed':alpha=np.exp(-((w+.10+.18*u*u)/.10)**2)*np.clip((1-ds)*5,0,1)*(sign*z>0)
        c=c*(1-alpha[:,None])+np.array([41,34,27])*alpha[:,None]
        if state!='closed':
            glint=np.exp(-((u+.25)**2/.03+(w-.21)**2/.05))*.70*alpha
            c=c*(1-glint[:,None])+np.array([244,232,207])*glint[:,None]
    return c

def tongue_mesh(jaw,s,b,angle,lick=False,nose_contact=None,headfield=None):
    # Samples are authored on the actual jaw centerline before rigid posing.
    start=jaw['x'][0]+5;tip=jaw['x'][0]-((1.8 if b=='maltipoo' else 1.1) if not lick else .4)
    t=np.linspace(0,1,90);x=start+(tip-start)*t
    y=np.interp(x,jaw['x'],jaw['y'])+jaw['thick']+.32
    pts=rotate(np.column_stack([x,y,np.zeros_like(x)]),jaw['hinge'],angle)
    if lick:
        nasal_y=float((s['features'][0][0][0,:,1].max()+s['features'][0][0][0,:,1].min())/2)
        contact=np.array([s['xs'][0],nasal_y,0]) if nose_contact is None else nose_contact
        w=model.smoothstep((t-.50)/.5)
        pts=pts*(1-w[:,None])+contact*w[:,None]
        if b=='maltipoo' and headfield is not None:
            xx=np.linspace(s['xs'][0]-3,jaw['hinge'][0],240)
            q=np.stack(np.broadcast_arrays(xx[None,:],pts[:,1,None],np.zeros((len(t),len(xx)))),axis=-1)
            inside=headfield(q)<0
            first=inside.argmax(1);front=xx[first]
            valid=inside.any(1)&(pts[:,1]>PARAMS[b]['lip_front']-.5)
            route=front-.30*(1-model.smoothstep((t-.92)/.08))
            pts[:,0]=np.where(valid,np.minimum(pts[:,0],route),pts[:,0])
            pts[-1]=contact
    else:pts[:,1]-=(1.4 if b=='maltipoo' else .8)*model.smoothstep((t-.6)/.4)
    width={'golden':1.65,'whippet':.95,'dachshund':1.2,'maltipoo':1.7}[b]
    theta=np.linspace(0,2*np.pi,49)
    taper=np.sqrt(np.clip(t/.07,0,1)*np.clip((1-t)/.12,0,1))
    v=pts[:,None,:]+np.stack(np.broadcast_arrays(np.zeros((90,49)),(.32*taper)[:,None]*np.cos(theta),(width*taper)[:,None]*np.sin(theta)),axis=-1)
    return v,np.array([202,117,122])

def oral_liner(s,jaw,angle):
    # Inner oral tissue bridges the fixed maxilla and articulated mandible.
    # It lies behind the lip rims and bows posteriorly into the mouth.
    f=s['facial_frame'];lip=f['face.upper_lip'];corner=f['face.L_mouth_corner']
    z=np.linspace(-3.4,3.4,81);u=z/3.4
    x=lip[0]+1.2+(corner[0]-lip[0]-1.6)*u*u
    y=oral_split(np.column_stack([x,x*0,z]),s)
    upper=np.column_stack([x,y,z]);lower=rotate(upper,jaw['hinge'],angle)
    t=np.linspace(0,1,49)
    v=upper[:,None,:]*(1-t[None,:,None])+lower[:,None,:]*t[None,:,None]
    v[:,:,0]+=.7*(1-u*u)[:,None]*np.sin(t*np.pi)
    return v,ORAL.copy()

def keep_outside(v,field,clearance=0.):
    v=v.copy()
    for _ in range(5):
        d=field(v);mask=d<clearance-1e-5
        if not np.any(mask):break
        q=v[mask];grad=[]
        for axis in np.eye(3)*.015:
            grad.append((field(q+axis)-field(q-axis))/.03)
        grad=np.stack(grad,axis=-1);grad/=np.maximum(np.linalg.norm(grad,axis=-1,keepdims=True),1e-6)
        v[mask]+=(clearance-d[mask])[:,None]*grad
    return v

def build(breeds=BREEDS):
    rigs,raw=inputs();heads={};evidence={}
    for b in breeds:
        s=sections(rigs[b],raw[b],b)
        v,f,n=extract(lambda q:head_field(q,s,rigs[b],b),[(s['xs'][0]-1,19),(-13,20),(-15,15)])
        jaw=jaw_mesh(s,rigs[b],b);states={}
        for state in STATES:
            angle=PARAMS[b]['angle']*(.5 if state=='mid' else .70 if state=='lick' else 1 if state in ['open','tongue'] else 0)
            jv=rotate(jaw['v'],jaw['hinge'],angle);jn=rotate(jaw['n'],np.zeros(3),angle)
            jc=np.tile(np.array(model.old.COLORS.get(b,([239,228,210],))[0]),(len(jv),1)).astype(float)
            if b=='maltipoo':jc[:]=MALTIPOO_COAT
            top=jaw['v'][:,1]-(np.interp(jaw['v'][:,0],jaw['x'],jaw['y']))
            oral=model.smoothstep((top-jaw['thick']*.3)/.3)
            if b=='maltipoo' and angle==0:oral*=0
            jc=jc*(1-oral[:,None])+ORAL*oral[:,None]
            grids=s['features'][1:]
            if b=='maltipoo' and angle>0:grids=grids+[oral_liner(s,jaw,angle)]
            if state in ['tongue','lick']:
                contact=v[np.argmin((v[:,0]-s['xs'][0])**2+v[:,2]**2)]
                tv,tc=tongue_mesh(jaw,s,b,angle,state=='lick',contact,lambda q:head_field(q,s,rigs[b],b))
                for _ in range(4):
                    tv=rotate(keep_outside(rotate(tv,jaw['hinge'],-angle),jaw['field']),jaw['hinge'],angle)
                    tv=keep_outside(tv,lambda q:head_field(q,s,rigs[b],b))
                grids=grids+[(tv,tc)]
            states[state]=dict(tri=[(v,f,n,colors(v,s,rigs[b],b,state)),(jv,jaw['f'],jn,jc)],grids=grids)
        heads[b]=states;evidence[b]=dict(sections=s,jaw=jaw,head=(v,f,n))
        print('Built',b,flush=True)
    return heads,evidence,rigs

def polys(mesh,angle,elev=0,ears=True):
    a,e=np.radians([angle,elev]);right=np.array([np.sin(a),0,np.cos(a)]);front=np.array([-np.cos(a)*np.cos(e),np.sin(e),np.sin(a)*np.cos(e)]);up=np.cross(front,right);view=np.column_stack([right,up,front])
    vv=[];cc=[];light=np.array([-.47,.70,.54])
    for v,f,n,c in mesh['tri']:
        verts=v@view;norm=n@view
        shade=.78+.22*np.maximum(0,norm@light)+.065*np.maximum(0,norm[:,1])
        rgba=c*shade[:,None]
        ix=f[:,[0,1,2,2]];vv.append(verts[ix]);cc.append(rgba[ix])
    grids=mesh['grids'] if ears else mesh['grids'][2:]
    if grids:
        v,c=model.polys(grids,angle,elev);vv.append(v);cc.append(c)
    return np.concatenate(vv)*1.7,np.concatenate(cc)
